## Parallelization

독립적인 여러 Node를 같은 단계에 연결하면 병렬로 실행할 수 있다.

~~~text
START ─┬→ legal_analysis     ─┐
       ├→ financial_analysis ─┼→ summarize → END
       └→ technical_analysis ─┘
~~~

각 분석 결과를 하나의 리스트로 모으려면 Annotated[list, operator.add] reducer를 사용한다. 병렬 Node의 완료 순서는 보장되지 않으므로 순서가 필요하면 결과에 식별자를 포함하고 후처리에서 정렬한다.

Voting 패턴은 여러 후보를 동시에 생성한 뒤 Judge LLM으로 최선의 답변을 선택하는 구조다.

## 실습 - 병렬 번역

~~~python
class State(TypedDict):
    text: str
    results: Annotated[list, operator.add]
    merge_text: str

def english_translator(state: State):
    response = llm.invoke("다음 문장을 영어로 번역해줘")

    return {"results": [("영어", response.text)]}

def japanese_translator(state: State):
    response = llm.invoke("다음 문장을 일본어로 번역해줘")

    return {"results": [("일본어", response.text)]}

def sapnish_translator(state: State):
    response = llm.invoke("다음 문장을 스페인어로 번역해줘")

    return {"results": [("스페인어", response.text)]}

def merge(state: State):
    result = ""
    for language, text in state["results"]:
        result += f"{language} - {text} \n"

    return {"merge_text": result}

builder = StateGraph(State)

builder.add_node('english', english_translator)
builder.add_node('japanese', japanese_translator)
builder.add_node('sapnish', sapnish_translator)
builder.add_node('merge', merge)

builder.add_edge(START, 'english')
builder.add_edge(START, 'japanese')
builder.add_edge(START, 'sapnish')

builder.add_edge(['english', 'japanese', 'sapnish'], 'merge')

graph = builder.compile()

result = graph.invoke({"text": "안녕하세요."})
print(result["merge_text"])
~~~

## Orchestrator-worker

Orchestrator가 작업 목록을 만들고, Send API로 작업마다 Worker를 동적으로 실행한 뒤 결과를 병합하는 패턴이다.

~~~text
START → Orchestrator ─┬→ Worker
                      ├→ Worker → 결과 종합 → END
                      └→ Worker
~~~

ReportState는 전체 요청·작업 목록·결과·최종 보고서를 저장하고, WorkerState는 Worker 하나가 처리할 입력만 저장한다. results에는 operator.add reducer를 적용하고, task_id나 order로 병렬 결과의 원래 순서를 복원한다.

Rate Limit이 걱정되면 max_concurrency로 동시에 실행할 Worker 수를 제한한다.

## 실습: 맞춤형 학습 자료 생성

~~~python
course_request = (
    "파이썬을 처음 배우는 비전공자를 위한 2시간 분량의 학습 자료를 만들어줘."
)

class Section(BaseModel):
    title: str = Field(description="목차")
    description: str = Field(description="학습할 내용")

class CoursePlan(BaseModel):
    sections: list[Section] = Field(description="맞춤형 학습 자료의 목차를 만드는 작업")

class CourseState(TypedDict):
    request: str
    sections: list[Section]
    results: Annotated[list, operator.add]
    course_materials: str

class SectionState(TypedDict):
    request: str
    section_id: int
    title: str
    description: str

def create_plan(state: CourseState):
    planner = llm.with_structured_output(CoursePlan)
    plan = planner.invoke(f"""
    다음 요청에 대한 학습 자료의 목차를 만들어줘.
    요청: {state['request']}
    """
    )
    return {"sections": plan.sections}

def assign_workers(state: CourseState):
    return [
        Send(
            "generate_section",
            {
                "request": state["request"],
                "section_id": section_id,
                "title": section.title,
                "description": section.description,
            }
        )
        for section_id, section in enumerate(state["sections"])
    ]


def generate_section(state: SectionState):
    result = llm.invoke(f"""
    다음 요청에 대한 학습 자료를 작성해줘.
    요청은 {state["request"]}, 제목과 학습 내용은 {state["title"]}, {state["description"]} 이야
    """
    )
    return {
        "results": [{
            "section_id": state["section_id"],
            "title": state["title"],
            "content": result.text,
        }]
    }


def make_course(state: CourseState):
    sorted_results = sorted(state["results"], key=lambda item: item["section_id"])
    all_results = "\n\n".join(
        f"### {item['title']}\n{item['content']}" for item in sorted_results
    )
    result = llm.invoke(
        f"다음 요청과 분석 결과를 바탕으로 하나의 학습 자료를 작성해줘.\n\n"
        f"요청: {state['request']}\n\n{all_results}"
    )
    return {"course_materials": result.text}
~~~

~~~python
graph_builder = StateGraph(CourseState)

graph_builder.add_node("create_plan", create_plan)
graph_builder.add_node("generate_section", generate_section)
graph_builder.add_node("make_course", make_course)

graph_builder.add_edge(START, "create_plan")
# Orchestrator가 만든 작업 수만큼 동적 Fan-out
graph_builder.add_conditional_edges("create_plan", assign_workers, ["generate_section"])
graph_builder.add_edge("generate_section", "make_course")
graph_builder.add_edge("make_course", END)

send_graph = graph_builder.compile()
display(Image(send_graph.get_graph().draw_mermaid_png()))
~~~

~~~python
request = "파이썬을 처음 배우는 비전공자를 위한 2시간 분량의 학습 자료를 만들어줘."
result = send_graph.invoke({"request": request})

print(result["course_materials"])
~~~

## 추가 실습

### 면접 문제 생성

~~~python
# job(직무), career(경력), stack(기술스택)

class Competency(BaseModel):
    keyword: str = Field(description="직무 역량 키워드 ex. 'cs', 'backend', 'langchain', '협업 능력' 등")
    key_point: str = Field(description="해당 직무 역량에서 중요하게 봐야하는 내용")

class InterviewPlan(BaseModel):
    competencies: list[Competency] = Field(description="직무 역량 키워드와 해당 역량에서 중요하게 봐야하는 내용")

class InterviewState(TypedDict):
    request: str
    competencies: list[Competency]
    results: Annotated[list, operator.add]
    questions: str

class CompetencyState(TypedDict):
    request: str
    keyword: str
    key_point: str
    order: int

def interview_plan(state: InterviewState):
    planner = llm.with_structured_output(InterviewPlan)
    plan = planner.invoke(f"""
    너는 IT 전문 기업의 면접관이다.
    사용자의 요청 내용(<request>)을 분석해 사용자가 가지고 있는 직무 역량을 작성해줘.
    
    <request>{state["request"]}</request>
    """)

    return {"competencies": plan.competencies}

def assign_workers(state: InterviewState):
    return [
        Send(
            "generate_interview_question",
            {
                "request": state["request"],
                "keyword": competency.keyword,
                "key_point": competency.key_point,
                "order": index,
            }
        )
        for index, competency in enumerate(state["competencies"])
    ]

def generate_interview_question(state: CompetencyState):
    result = llm.invoke(f"""
    다음 <request>에 대한 면접 질문을 아래를 참고해 작성해줘.
    - 키워드는 <keyword>, 각 키워드의 역량에서 중요하게 봐야하는 내용은 <key_point>
    - 키워드와 면접 질문만 작성할 것.
    - 마크다운으로 작성할것.
    
    <request>{state["request"]}</request>
    <keyword>{state["keyword"]}</keyword>
    <key_point>{state["key_point"]}</key_point>
    """)

    return {
        "results": [{
            "keyword": state["keyword"],
            "content": result.text,
            'order': state['order'],
        }]
    }

def mearge_results(state: InterviewState):
    results = state['results']
    results.sort(key=lambda x : x['order'])

    questions = "\n\n".join([
        f"""
    keyword : {result['keyword']} \n 
    content : {result['content']}
    """ 
    for result in results])

    return {'questions' : questions}    

builder = StateGraph(InterviewState)

builder.add_node("interview_plan", interview_plan)
builder.add_node("generate_interview_question", generate_interview_question)
builder.add_node("merge_results", mearge_results)

builder.add_edge(START, "interview_plan")
builder.add_conditional_edges("interview_plan", assign_workers, ["generate_interview_question"])
builder.add_edge("generate_interview_question", "merge_results")
builder.add_edge("merge_results", END)

graph = builder.compile()
display(Image(graph.get_graph().draw_mermaid_png()))
~~~

"Kotlin·Spring Boot를 기반으로 서버 개발과 외부 API 연동, 데이터 처리 및 성능 개선을 수행했습니다. "

print(result["questions"])
