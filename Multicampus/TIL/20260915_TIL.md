## Human-in-the-Loop

Human-in-the-Loop(HITL)는 결제, 게시글 발행처럼 중요한 판단이 필요한 지점에서 그래프를 중단하고 사람의 결정을 받은 뒤 재개하는 방식이다.

~~~text
AI 작업 수행 → interrupt → 승인·거절·추가 입력 → graph 재개
~~~

## 실습: 게시글 발행 승인하기

~~~python
@tool
def publish_post(title: str, content: str) -> str:
    """게시글을 발행한다.
    
    Args:
        title: 제목
        content: 본문
    """
    # TODO: 게시글 발행 전 승인을 받는 HITL 흐름을 구현하세요.
    decision = interrupt(
        {
            "question": "이 게시글을 발행 할까요?",
            "tool": "publish_post",
            "args": {"title": title, "content": content},
        }
    )

    if decision["action"] == "reject":
        reason = decision.get("reason", "사유 없음")
        return f"게시글 발행이 취소되었습니다. 사유: {reason}"

    return f"제목: {title} \n 본문: {content} \n 게시글 발행 완료."
~~~

class State(TypedDict):

tools = [publish_post]
llm = ChatGoogleGenerativeAI(model=MODEL_NAME).bind_tools(tools)

## 추가 실습: 필요한 정보 질문하기

~~~python
# 요청 -> 질문 -> 답변 -> 충족시 요청 수행
class State(TypedDict):
    request: str
    answers: Annotated[list, operator.add]
    question: str
    options: list[str]
    result: str
    is_sufficient: bool


class Question(BaseModel):
    question: str
    options: list[str]


llm = ChatGoogleGenerativeAI(model=MODEL_NAME)


def analzye_user_request(state: State):
    request = state["request"]
    answers = state.get("answers", None)

    result = llm.invoke(f"""
    사용자의 요청을 수행하는 데 반드시 필요한 정보가 현재 요청에 포함되어 있는지 확인 후 "충분"/"불충분"으로 판단해줘.
    - "충분" or "불충분"으로만 답변
    
    사용자 요청: {request}
    저장된 정보: {answers}
    """)

    print(result.text)

    if result.text == "충분":
        return {"is_sufficient": True}
    else:
        return {"is_sufficient": False}


def route(state: State):
    return "end" if state["is_sufficient"] else "continue"


def generate_question(state: State):
    request = state["request"]
    answers = state.get("answers", None)
    print(answers)
    question_agent = llm.with_structured_output(Question)
    result = question_agent.invoke(f"""
    사용자의 요청을 처리하기 위해 필요한 정보 요청을 객관식 질문 형태로 생성해줘.
    - 질문은 간단한 형식으로
    - 이미 질문하고 사용자가 답변한 내용과 비슷한 내용은 질문하지 않는다.

    사용자 요청: {request}
    저장된 정보: {answers}
    """)

    return {"question": result.question, "options": result.options}


def ask_question(state: State):
    decision = interrupt(
        {
            "question": state["question"],
            "options": state["options"],
        }
    )

    return {"answers": [(state["question"], decision["answer"])]}


def execute(state: State):
    request = state["request"]
    answers = state["answers"]

    result = llm.invoke(f"""
        사용자 요청: {request}
        저장된 정보: {answers}

        위 정보를 기반으로 사용자의 요청을 수행해줘.
    """)

    return {"result": result.text}


builder = StateGraph(State)
builder.add_node("analyze", analzye_user_request)
builder.add_node("generate_question", generate_question)
builder.add_node("ask_question", ask_question)
builder.add_node("execute", execute)

builder.add_edge(START, "analyze")
builder.add_conditional_edges(
    "analyze", route, {"continue": "generate_question", "end": "execute"}
)
builder.add_edge("generate_question", "ask_question")
builder.add_edge("ask_question", "analyze")
builder.add_edge("execute", END)

graph = builder.compile(checkpointer=MemorySaver())
display(Image(graph.get_graph().draw_mermaid_png()))
~~~
